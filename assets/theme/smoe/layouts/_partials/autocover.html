{{- /* 自动封面：uploads/background 下的 .avif 按文件名排序，按文章在列表中的位置轮换（与旧版一致）。 */ -}} {{- $covers := slice -}} {{- if os.FileExists "uploads/background" -}} {{- range os.ReadDir "uploads/background" -}} {{- if and (not .IsDir) (eq (lower (path.Ext .Name)) ".avif") }}{{ $covers = $covers | append .Name }}{{ end -}} {{- end -}} {{- end -}} {{- $url := "" -}} {{- with $covers -}} {{- range $i, $p := (where site.RegularPages "Section" "archives").ByDate.Reverse -}} {{- if eq $p $ }}{{ $url = printf "/usr/uploads/background/%s" (index $covers (mod $i (len $covers))) }}{{ end -}} {{- end -}} {{- end -}} {{- return $url -}}
